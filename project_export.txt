'use strict';

const SCHEMA_VERSION = 2;
const RULESET_IDS = Object.freeze(['ads', 'trackers', 'security']);
const ALLOW_RULE_BASE = 100000;
const ALLOW_RULE_MAX = 4500;
const EXTENSION_ORIGIN = chrome.runtime.getURL('');

const DEFAULTS = Object.freeze({
  enabled: true,
  cosmetic: true,
  allowlist: [],
  schemaVersion: SCHEMA_VERSION
});

const ALLOW_RULE_RESOURCE_TYPES = Object.freeze([
  'main_frame', 'sub_frame', 'stylesheet', 'script', 'image', 'font',
  'object', 'xmlhttprequest', 'ping', 'media', 'websocket', 'other'
]);

/* ---------- serialized operation queues ---------- */
function createQueue(label) {
  let tail = Promise.resolve();
  return function enqueue(operation) {
    const next = tail.then(operation);
    tail = next.catch(error => console.error(`[BMZ] ${label} failed`, error));
    return next;
  };
}

const runRuleset = createQueue('ruleset operation');
const runAllowlist = createQueue('allowlist operation');
const runState = createQueue('state operation');
const runCounts = createQueue('counter operation');

/* ---------- helpers ---------- */
function getToday() {
  return new Date().toISOString().split('T')[0];
}

function normalizeHost(value) {
  if (typeof value !== 'string') return null;
  const input = value.trim();
  if (!input || /[\s/@*\\?#]/.test(input)) return null;
  try {
    const url = new URL(input.includes('://') ? input : `https://${input}`);
    if (!['http:', 'https:'].includes(url.protocol)) return null;
    if (url.username || url.password) return null;
    if (url.pathname !== '/' || url.search || url.hash) return null;
    const host = url.hostname.toLowerCase().replace(/\.$/, '').replace(/^www\./, '');
    if (!host || host.length > 253 || !host.includes('.') || !/^[a-z0-9.-]+$/.test(host) || host.startsWith('.') || host.endsWith('.') || host.includes('..')) {
      return null;
    }
    return host;
  } catch {
    return null;
  }
}

function normalizeAllowlist(value) {
  if (!Array.isArray(value)) return [];
  return [...new Set(value.map(normalizeHost).filter(Boolean))].slice(0, ALLOW_RULE_MAX);
}

/* ---------- State ---------- */
async function readState() {
  const stored = await chrome.storage.local.get(DEFAULTS);
  const state = {
    enabled: typeof stored.enabled === 'boolean' ? stored.enabled : DEFAULTS.enabled,
    cosmetic: typeof stored.cosmetic === 'boolean' ? stored.cosmetic : DEFAULTS.cosmetic,
    allowlist: normalizeAllowlist(stored.allowlist),
    schemaVersion: SCHEMA_VERSION
  };
  const updates = {};
  if (stored.schemaVersion !== SCHEMA_VERSION) updates.schemaVersion = SCHEMA_VERSION;
  if (!Array.isArray(stored.allowlist) || stored.allowlist.length !== state.allowlist.length) updates.allowlist = state.allowlist;
  if (Object.keys(updates).length) await chrome.storage.local.set(updates);
  return state;
}

/* ---------- Rulesets ---------- */
async function syncBadge(enabled) {
  try {
    if (enabled) {
      await chrome.action.setBadgeText({ text: '' });
    } else {
      await chrome.action.setBadgeBackgroundColor({ color: '#c2410c' });
      await chrome.action.setBadgeText({ text: 'OFF' });
    }
  } catch (e) {}
}

async function syncRulesets() {
  const { enabled } = await readState();
  const currentlyEnabled = await chrome.declarativeNetRequest.getEnabledRulesets();
  const wanted = enabled ? [...RULESET_IDS] : [];
  const enableRulesetIds = wanted.filter(id => !currentlyEnabled.includes(id));
  const disableRulesetIds = currentlyEnabled.filter(id => !wanted.includes(id));

  if (enableRulesetIds.length || disableRulesetIds.length) {
    await chrome.declarativeNetRequest.updateEnabledRulesets({ enableRulesetIds, disableRulesetIds });
  }
  await syncBadge(enabled);
}

async function syncAllowlistRules() {
  const { allowlist } = await chrome.storage.local.get({ allowlist: [] });
  const clean = normalizeAllowlist(allowlist);
  const existing = await chrome.declarativeNetRequest.getDynamicRules();
  const removeRuleIds = existing.filter(rule => rule.id >= ALLOW_RULE_BASE && rule.id < ALLOW_RULE_BASE + ALLOW_RULE_MAX).map(rule => rule.id);

  const addRules = clean.map((host, index) => ({
    id: ALLOW_RULE_BASE + index,
    priority: 100000,
    action: { type: 'allow' },
    condition: { initiatorDomains: [host], resourceTypes: [...ALLOW_RULE_RESOURCE_TYPES] }
  }));

  if (!removeRuleIds.length && !addRules.length) return;
  await chrome.declarativeNetRequest.updateDynamicRules({ removeRuleIds, addRules });
}

/* ---------- Cosmetic Counting (DOM) ---------- */
// Bumps three counters in one atomic write:
//   domDaily   — session-scoped total, resets at midnight
//   dailyCount — same as domDaily, kept for popup "Today" label
//   tabCounts  — per-tab count, resets on navigation, cleared on close
function recordCosmetic(count, tabId) {
  return runCounts(async () => {
    const today = getToday();
    const data = await chrome.storage.local.get({
      domDaily: 0, dailyCount: 0, dailyDate: today, tabCounts: {}
    });
    let domDaily = data.domDaily;
    let dailyCount = data.dailyCount;
    if (data.dailyDate !== today) {
      domDaily = 0;
      dailyCount = 0;
    }
    const tabCounts = data.tabCounts;
    if (typeof tabId === 'number' && tabId > 0) {
      tabCounts[tabId] = (tabCounts[tabId] || 0) + count;
    }
    await chrome.storage.local.set({
      domDaily: domDaily + count,
      dailyCount: dailyCount + count,
      dailyDate: today,
      tabCounts
    });
  });
}

// Clear tab count on close so tabCounts does not grow forever.
chrome.tabs.onRemoved.addListener(async (tabId) => {
  const { tabCounts = {} } = await chrome.storage.local.get('tabCounts');
  if (tabCounts[tabId] !== undefined) {
    delete tabCounts[tabId];
    await chrome.storage.local.set({ tabCounts });
  }
});

// Reset tab count when the user navigates or reloads.
chrome.tabs.onUpdated.addListener(async (tabId, changeInfo) => {
  if (changeInfo.status === 'loading' && changeInfo.url) {
    const { tabCounts = {} } = await chrome.storage.local.get('tabCounts');
    if (tabCounts[tabId] !== undefined) {
      tabCounts[tabId] = 0;
      await chrome.storage.local.set({ tabCounts });
    }
  }
});

chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
  const type = message?.type;
  if (typeof type !== 'string') return false;

  (async () => {
    try {
      switch (type) {
        case 'ping':
          sendResponse({ ok: true });
          break;

        case 'getState': {
          const [state, activeRulesets, data] = await Promise.all([
            readState(),
            chrome.declarativeNetRequest.getEnabledRulesets(),
            chrome.storage.local.get({ dailyCount: 0, tabCounts: {}, domDaily: 0 })
          ]);

          const tabId = message.tabId || (sender.tab ? sender.tab.id : -1);
          const pageCount = data.tabCounts[tabId] || 0;

          sendResponse({
            enabled: state.enabled,
            cosmetic: state.cosmetic,
            allowlist: state.allowlist,
            activeRulesets,
            counts: { page: pageCount, daily: data.dailyCount, dom: data.domDaily }
          });
          break;
        }

        case 'setEnabled': {
          if (!sender?.url?.startsWith(EXTENSION_ORIGIN)) return sendResponse({ ok: false });
          await chrome.storage.local.set({ enabled: message.value });
          await runRuleset(syncRulesets);
          sendResponse({ ok: true });
          break;
        }

        case 'addAllow': {
          if (!sender?.url?.startsWith(EXTENSION_ORIGIN)) return sendResponse({ ok: false });
          const host = normalizeHost(message.host);
          if (!host) return sendResponse({ ok: false });

          await runState(async () => {
            const { allowlist = [] } = await chrome.storage.local.get({ allowlist: [] });
            const current = normalizeAllowlist(allowlist);
            if (!current.includes(host) && current.length < ALLOW_RULE_MAX) {
              await chrome.storage.local.set({ allowlist: [...current, host] });
            }
          });
          await runAllowlist(syncAllowlistRules);
          sendResponse({ ok: true });
          break;
        }

        case 'removeAllow': {
          if (!sender?.url?.startsWith(EXTENSION_ORIGIN)) return sendResponse({ ok: false });
          const host = normalizeHost(message.host);
          if (!host) return sendResponse({ ok: false });

          await runState(async () => {
            const { allowlist = [] } = await chrome.storage.local.get({ allowlist: [] });
            const current = normalizeAllowlist(allowlist);
            await chrome.storage.local.set({ allowlist: current.filter(entry => entry !== host) });
          });
          await runAllowlist(syncAllowlistRules);
          sendResponse({ ok: true });
          break;
        }

        case 'cosmeticHidden': {
          if (!sender?.tab) return sendResponse({ ok: false });
          await recordCosmetic(message.count, sender.tab.id);
          sendResponse({ ok: true });
          break;
        }

        case 'recordSlippedAd': {
          if (!sender?.tab) return sendResponse({ ok: false });
          const rec = message.record;
          if (!rec || rec.__bmz !== 'adSlipped') return sendResponse({ ok: false });

          await runCounts(async () => {
            const { slippedAds = [] } = await chrome.storage.local.get({ slippedAds: [] });
            slippedAds.push({
              ts: rec.ts || Date.now(),
              url: typeof rec.url === 'string' ? rec.url.slice(0, 2048) : '',
              tabId: sender.tab.id,
              signals: rec.signals && typeof rec.signals === 'object' ? rec.signals : {}
            });
            await chrome.storage.local.set({ slippedAds: slippedAds.slice(-200) });
          });

          sendResponse({ ok: true });
          break;
        }

        case 'getSlippedAds': {
          const { slippedAds = [] } = await chrome.storage.local.get({ slippedAds: [] });
          sendResponse({ ok: true, slippedAds });
          break;
        }

        default:
          sendResponse({ ok: false, error: 'unknown' });
      }
    } catch (e) {
      console.error('[BMZ] message handler failed', e);
      try { sendResponse({ ok: false }); } catch {}
    }
  })();

  return true;
});

  let adSeekAttempts = 0;         // seek budget, reset for EVERY new ad in a pod
  let sped = null;                // saved {muted, rate} while we fast-forward an ad

  function isEnabled() {
    return document.documentElement.getAttribute('data-bmz-enabled') !== '0';
  }

  function ensureAntiAdblockStyle() {
    if (document.getElementById('bmz-anti-adblock-css')) return;
    const style = document.createElement('style');
    style.id = 'bmz-anti-adblock-css';
    style.textContent = `
      ytd-enforcement-message-view-model { display: none !important; }
      ytd-popup-container:has(> ytd-enforcement-message-view-model) { display: none !important; }
    `;
    document.documentElement.appendChild(style);
  }

  function getPlayer() { return document.querySelector('.html5-video-player'); }
  function getVideo()  { return document.querySelector('video.html5-main-video'); }

  function isAdShowing() {
    const p = getPlayer();
    return !!(p && p.classList.contains('ad-showing'));
  }

  function isAdActive() {
    const player = getPlayer();
    if (player && player.classList.contains('ad-showing')) return true;
    for (const badge of document.querySelectorAll(AD_BADGE_SELECTORS)) {
      if (badge.offsetParent !== null || badge.getClientRects().length) return true;
    }
    if (document.querySelector('.ytp-ad-duration-remaining')) return true;
    if (player && /(Sponsored|Ad · \d+ of \d+)/i.test(player.textContent || '')) return true;
    return false;
  }

  function hideOverlayAds() {
    for (const el of document.querySelectorAll(OVERLAY_SELECTORS)) {
      if (el.style.getPropertyValue('display') !== 'none') {
        try { el.style.setProperty('display', 'none', 'important'); } catch {}
      }
    }
  }

  function dismissEnforcementPopup() {
    const popups = document.querySelectorAll(ENFORCEMENT_SELECTOR);
    if (!popups.length) return false;
    const toRemove = new Set();
    for (const popup of popups) {
      const dialog = popup.closest('tp-yt-paper-dialog');
      const container = popup.closest('ytd-popup-container');
      if (dialog) toRemove.add(dialog);
      if (container) toRemove.add(container);
      try { popup.remove(); } catch {}
    }
    for (const node of toRemove) {
      const prev = node.previousElementSibling;
      if (prev && prev.tagName === 'TP-YT-IRON-OVERLAY-BACKDROP') {
        try { prev.remove(); } catch {}
      }
      try { node.remove(); } catch {}
    }
    const video = getVideo();
    if (video && video.paused) try { video.play().catch(() => {}); } catch {}
    return true;
  }

  // Click regardless of visibility: .click() works on hidden elements, and a
  // visibility check can wrongly skip a button inside a collapsed container.
  function clickSkipButtons() {
    for (const btn of document.querySelectorAll(SKIP_SELECTORS)) {
      if (btn.disabled) continue;
      try { btn.click(); } catch {}
    }
  }

  // Runs only while the player has .ad-showing, so it can never touch real content.
  // Strategy: mute + fast-forward immediately (works even if seeks are ignored or
  // duration is NaN), AND try to seek to the end. Budget resets for every new ad.
  function skipAd(video) {
    if (!video) return;

    const dur = video.duration;
    const key = (video.currentSrc || '') + '|' + (Number.isFinite(dur) ? dur.toFixed(2) : 'nan');
    if (key !== adKey) { adKey = key; adSeekAttempts = 0; }

    if (!sped) sped = { muted: video.muted, rate: video.playbackRate };
    try { video.muted = true; } catch {}
    try { if (video.playbackRate !== FAST_RATE) video.playbackRate = FAST_RATE; } catch {}
    if (video.paused) try { video.play().catch(() => {}); } catch {}

    if (Number.isFinite(dur) && dur > 0 &&
        Number.isFinite(video.currentTime) && video.currentTime < dur - 0.3 &&
        adSeekAttempts < MAX_SEEK_ATTEMPTS) {
      adSeekAttempts++;
      try { video.currentTime = dur; } catch {}
    }
  }

  function restoreAfterAd() {
    if (!sped) return;
    const v = getVideo();
    if (v) {
      try { v.muted = sped.muted; } catch {}
      try { v.playbackRate = sped.rate || 1; } catch {}
    }
    sped = null;
    adKey = '';
    adSeekAttempts = 0;
  }

  function tick() {
    if (!isEnabled()) return;

    ensureAntiAdblockStyle();
    dismissEnforcementPopup();
    hideOverlayAds();

    const ad = isAdActive();
    const video = getVideo();

    if (ad) {
      clickSkipButtons();
      if (isAdShowing()) skipAd(video);
      log('ad', { showing: isAdShowing(), dur: video && video.duration, cur: video && video.currentTime, attempts: adSeekAttempts });

      if (!adReported) {
        adReported = true;
        setTimeout(() => {
          if (!isAdActive()) { adReported = false; return; }
          const badge = document.querySelector(AD_BADGE_SELECTORS);
          try {
            window.postMessage({
              __bmz: 'adSlipped',
              token: SECRET_TOKEN,
              url: location.href,
              signals: {
                adShowing: isAdShowing(),
                badgeText: badge ? (badge.textContent || '').trim().slice(0, 200) : '',
                skipButtonPresent: document.querySelectorAll(SKIP_SELECTORS).length > 0,
                videoDuration: video ? video.duration : null
              },
              ts: Date.now()
            }, '*');
          } catch {}
          adReported = false;
        }, 1500);
      }
    } else {
      restoreAfterAd();
    }
  }

  // --- Triggers: poll (never paused in background tabs) + events + DOM observer ---
  setInterval(tick, 250);

  let fastTicks = 0;
  const fastTimer = setInterval(() => {
    tick();
    if (++fastTicks >= 24) clearInterval(fastTimer);
  }, 125);

  document.addEventListener('visibilitychange', () => { tick(); });

  // Media events don't bubble, but they can be captured at the document.
  for (const ev of ['loadstart', 'loadedmetadata', 'durationchange', 'playing', 'emptied']) {
    document.addEventListener(ev, (e) => {
      if (e.target && e.target.tagName === 'VIDEO') tick();
    }, true);
  }

  // React the instant the player flips into/out of ad mode.
  let observedPlayer = null;
  const mo = new MutationObserver(() => tick());
  function attachObserver() {
    const p = getPlayer();
    if (p && p !== observedPlayer) {
      mo.disconnect();
      observedPlayer = p;
      mo.observe(p, { attributes: true, attributeFilter: ['class'] });
    }
  }
  attachObserver();
  setInterval(attachObserver, 1000);

  document.addEventListener('yt-navigate-finish', () => {
    adReported = false;
    attachObserver();
    let n = 0;
    const t = setInterval(() => { tick(); if (++n >= 24) clearInterval(t); }, 125);
  });

  let lastHref = location.href;
  setInterval(() => {
    if (location.href === lastHref) return;
    lastHref = location.href;
    adReported = false;
    adKey = '';
    adSeekAttempts = 0;
    attachObserver();
    let n = 0;
    const t = setInterval(() => { tick(); if (++n >= 24) clearInterval(t); }, 125);
  }, 500);
})();

# Filter Sources

## Current Release (v0.1.0)

**No third-party filter-list data is bundled.** All rules in `rules/*.json` are intended to be hand-authored for this project and distributed under the same MIT license as the rest of the source tree.

This is a deliberate decision for the initial release:

- **Licensing clarity:** Avoid importing third-party list data before its licensing and redistribution requirements have been reviewed.
- **Reproducible builds:** Avoid depending on an unpinned conversion tool whose output or interpretation may change between builds.
- **Auditability:** Keep the shipped rulesets small, understandable, and straightforward to review.

These statements describe the intended policy. The actual rule files and build configuration must be checked to confirm compliance.

## Requirements for Adding External Lists

Any third-party filter list must be documented in this file **before it is imported**. Each source record must include:

| Field | Description |
|---|---|
| `id` | Stable identifier, such as `easylist-ads`. |
| `source_url` | Canonical HTTPS URL for the upstream source. |
| `license` | SPDX identifier and any applicable additional terms. |
| `attribution` | Exact attribution required by the applicable license. |
| `conversion` | Converter name, version, and configuration used to generate DNR rules. |
| `last_fetched` | ISO 8601 date of the last successful build-time fetch. |
| `sha256` | SHA-256 checksum of the upstream file fetched for that build. |
| `expected_rule_count` | Expected output rule count and an acceptable change threshold. |
| `redistributable` | Whether the source and resulting rule data may legally be redistributed. |

The build should fail if required metadata is missing, licensing is unresolved, or generated output exceeds configured validation limits.

Upstream changes should be reviewed explicitly. A rule-count change greater than 10% should trigger a build failure pending review rather than being silently accepted.

A metadata record alone does not establish that a source or its converted output is legally redistributable. Licensing must be evaluated for the actual source, applicable terms, and intended distribution.

## Sources Considered but Not Bundled (v0.1.0)

The following sources are potential candidates for a future build-time integration. **None are declared as active dependencies by this document.** Their current licensing and redistribution terms must be verified before adoption.

- **EasyList** — https://easylist.to/easylist/easylist.txt
- **EasyPrivacy** — distributed through the EasyList project.
- **AdGuard Base** — https://filters.adtidy.org/extension/ublock/filters/2.txt
- **AdGuard Annoyances** — available through AdGuard's filter distribution infrastructure.
- **URLhaus host file** — https://urlhaus.abuse.ch/downloads/hostfile/
- **Spam404** — https://raw.githubusercontent.com/Spam404/lists/master/main-blacklist.txt
- **uBlock Origin filter lists** — review applicable licensing, attribution, branding, and redistribution requirements before use.

The source names and URLs above are candidates only. They must not be interpreted as confirmation of current availability, license compatibility, or permission to redistribute converted data.

Blocking unreviewed list imports is intentional. Adding a third-party source is a policy and supply-chain change that requires review, not merely a code change.

## Build-Time Import Policy

If external sources are approved in a future release, the build pipeline should:

1. Fetch only from explicitly approved HTTPS origins.
2. Pin converter versions and configuration.
3. Record source checksums and retrieval dates.
4. Validate input formats before conversion.
5. Validate generated DNR JSON against Chrome's supported schema and rule constraints.
6. Enforce per-ruleset rule-count and file-size limits.
7. Compare generated output against expected bounds and flag unexpected changes.
8. Preserve license notices and required attribution in the distributed package.
9. Produce reproducible, reviewable build artifacts.
10. Fail closed when source integrity, validation, or licensing checks fail.

## Local Runtime Updates

The v0.1.0 policy is **no runtime filter-list fetching or updating**.

Any future updater must undergo a separate security review and satisfy these requirements:

1. Fetch only from an explicit HTTPS origin allowlist.
2. Validate downloaded data, schema, rule counts, and byte-size limits before applying changes.
3. Apply updates through supported Chrome Declarative Net Request APIs.
4. Preserve the last known-good ruleset until the replacement has passed validation.
5. Roll back safely if applying or verifying the replacement fails.
6. Never download or execute JavaScript as filter-list content.
7. Keep remote data strictly separate from executable extension code.
8. Document update behavior, integrity controls, and failure handling before release.

**Release guarantee:** Filter-list data is packaged with the extension. No remote filter-list update mechanism is part of the intended v0.1.0 design.

Verify the manifest, build scripts, generated rules, and runtime code before presenting these policy statements as confirmed implementation guarantees.

# Privacy Policy for BuildMintZ Shield

**Last Updated: October 22, 2026**

BuildMintZ Shield is a local-only browser extension. We believe your browsing history and data belong to you.

## Data Collection
**We do not collect, store, transmit, or sell any personal data.** 
BuildMintZ Shield does not contain any analytics, telemetry, tracking pixels, or remote servers. 

## Local Storage
All settings, including your global pause state, cosmetic filtering toggle, per-site allowlist, and ad-slip diagnostics, are stored locally on your device using the browser's `chrome.storage.local` API. This data never leaves your computer.

## Network Requests
The extension uses Chrome's native `declarativeNetRequest` API to block requests to known ad, tracker, and malicious domains. It does not read the content of your network traffic. 

## Permissions
- **storage:** Required to save your settings and allowlist locally.
- **activeTab:** Required to read the current URL so the popup can display the correct site exception button.
- **declarativeNetRequest:** Required to block network requests to known ad and tracker domains.
- **host_permissions (http/https):** Required so the extension can apply network rules and cosmetic filtering across websites.
- **Video-player content scripts:** Required to detect and skip skippable video ads and dismiss client-side anti-adblock prompts on supported video sites.

## Contact
If you have any questions about this privacy policy, please open an issue on our GitHub repository.

# Security model

## Protected against

- Network requests to **85 known ad domains, 68 tracker domains, and 21
  confirmed malicious/coercive-redirect domains** — 171 static rules total
  across three rulesets. These counts are machine-verified against
  `rules/*.json` by `npm run validate`; a mismatch fails the build.
  Filtering is performed by Chrome's `declarativeNetRequest` engine in the
  browser process, before the response is delivered to the requesting
  context: top-level document, sub-frame, script, stylesheet, image, font,
  XHR/fetch, ping, media, WebSocket, or other. The page still executes its
  own scripts; only the blocked request's response is withheld.
- Common cosmetic ad containers (Google Ads slots, GAM iframes, Taboola,
  Outbrain, sponsored test IDs) hidden by a scoped stylesheet plus a
  `MutationObserver` that catches dynamically inserted nodes.
- Popunder, paid-link shortener, and crypto-miner domains blocked including
  for `main_frame` navigations. **This protection depends on the `security`
  ruleset being enabled.** With `security` off and `ads` on, those domains
  remain blocked as sub-resources but top-level navigation to them is
  allowed. The default configuration enables all three rulesets.

## Not protected against

- **Unknown or newly registered** phishing and malware domains. The bundled
  security list is hand-curated, not a reputation feed, and does not update
  itself.
- Malicious ads served from a site's own origin, or from a trusted CDN the
  site depends on. These cannot be distinguished from content by network
  rules or by DOM selectors.
- Same-origin video-player ads. Delivery from the site's own origin makes
  them unblockable by network rules; where the player exposes a skip
  control, the content script uses it. Server-stitched ads cannot be
  skipped by any client-side technique; the player continues to work
  normally.
- First-party sponsored posts. Server-rendered content carries no network
  signal an extension can block.
- **Ads rendered inside iframes.** The content script is declared with
  `all_frames: false`, so cosmetic hiding runs only in the top-level frame.
  Network rules still block ad-network iframes at the request layer, but a
  same-origin or unblocked third-party iframe that renders an ad will not be
  cosmetically hidden.
- Drive-by downloads, exploit chains, or browser zero-days.
- Any threat outside the browser (email, DNS, OS).

## Design principles

- **No remote code execution.** No `eval`, no dynamically fetched scripts,
  no runtime rule conversion. All rules ship as JSON data inside the
  extension package.
- **Local-only data.** The extension does not intentionally transmit user
  data or browsing history to BuildMintZ or any third party. Settings live
  in `chrome.storage.local`; session counters live in
  `chrome.storage.session` and are cleared by the browser. Chrome itself
  makes its own network calls — extension updates, Safe Browsing lookups,
  crash reporting — under its own policies, independent of this extension.
- **DNR over request interception.** Blocking is handled by Chrome's native
  network engine. No per-request JavaScript, no page-world hooks, no
  `window.open` patching.
- **Stateless background worker.** The Manifest V3 service worker holds no
  load-bearing in-memory state. Counters are persisted to
  `chrome.storage.session` on every report so worker eviction cannot lose
  data. Every listener is registered synchronously at the top level.
- **Least privilege.** The manifest requests `storage`, `activeTab`, and
  `declarativeNetRequest` only. `declarativeNetRequestFeedback` is omitted:
  nothing in the shipped code calls `getMatchedRules()` or
  `onRuleMatchedDebug`, and the counters are derived from the DOM, not from
  matched-rule events.
- **Fail-open on cosmetic, fail-safe on network.** If the content script
  throws, the page is unaffected; ads stay visible but the site works. If a
  network rule is wrong, the user can allowlist the site from the popup
  without reinstalling.

## Site exceptions

A host in `allowlist`:

- Receives a dynamic DNR `allow` rule with `priority: 100000`, overriding
  all bundled block rules (priority 1 in `ads`/`trackers`, 100 in
  `security`) for requests **initiated by** that host. Matching uses
  `initiatorDomains`, so the exception applies to requests the site makes,
  not to third-party iframes layered on top of it.
- Is skipped by the content script's cosmetic filter.
- Is stored locally and never transmitted.

There is no wildcard support. `example.com` also matches `sub.example.com`
(initiator-domain matching is subtree-aware) but does not match
`notexample.com`.

Allow rules occupy the reserved ID range `100000`–`104499`. The cap of 4,500
entries keeps the extension inside Chrome's 30,000 dynamic-rule budget and
Firefox's 5,000-per-ruleset quota with headroom.

## Counting

"Elements hidden" is an **approximate** count of DOM nodes matching our
cosmetic selectors that were added during the session. It:

- Includes nodes that were later removed or hidden by the site's own CSS.
  The content script counts a node when it observes it; it does not verify
  that the node remained hidden.
- Does not count network-blocked requests. `onRuleMatchedDebug` fires only
  for unpacked extensions and cannot back a shipped product's counter.
- Is persisted to `chrome.storage.session`, so it survives service worker
  eviction. It resets when the browser closes.

It is not proof that ads were blocked. It is a session-scoped signal that
the cosmetic layer is doing work.

## Known trade-offs

- **Blocking `googletagmanager.com/gtag/js` and `gtm.js`** also breaks consent
  management platforms and A/B testing tools that are delivered through GTM.
  Some sites will show a cookie banner that never resolves. Allowlisting the
  site is the intended remedy.
- **Blocking `facebook.com/tr`** can break conversion tracking used by
  e-commerce checkout flows; the purchase still completes, but the merchant
  loses attribution. This is the intended behaviour of a tracker blocker.
- **Blocking `bat.bing.com`** can break Microsoft Advertising conversion
  reporting with the same caveat.

## Incident response

If a page breaks:

1. Use the popup to allowlist that site.
2. Report the URL and which ruleset is responsible (DevTools → Network →
   look for `blockedbyclient` or `ERR_BLOCKED_BY_CLIENT`, then match the
   request host against `rules/*.json`).
3. Do not add broad allow rules. Allowlisting one host is the intended
   escape hatch.

## Consistency requirements

This document describes intent. It becomes a verified claim only when these
files agree with it:

- `manifest.json` — permissions, DNR ruleset declarations, content script
  matches and `all_frames`
- `background.js` — allowlist mechanism (dynamic rules, priority 100000,
  reserved ID band), stateless counter implementation
- `content/content.js` — cosmetic count source
- `rules/ads.json`, `rules/trackers.json`, `rules/security.json` — rule
  counts, `main_frame` presence, resource types

`npm run validate` enforces the rule counts quoted at the top of this file
against the actual JSON, rejects duplicate `urlFilter` values within a
ruleset, rejects unknown condition/action keys, cross-checks every element
ID referenced by `popup.js` and `options.js` against its HTML, and verifies
that `package.json` and `manifest.json` versions match.

Chrome 120+ allows up to 100 declared static rulesets (50 enabled) with a
guaranteed 30,000 static rules; this project caps itself at 25,000 total and
10,000 per ruleset so it stays comfortably inside both Chrome's guarantee and
Firefox's stricter installation-time behaviour.

# Notices and filter provenance

## v0.1.0

This release contains **hand-authored domain rules only**, in
`rules/ads.json`, `rules/trackers.json`, and `rules/security.json`. It does
**not** bundle EasyList, EasyPrivacy, AdGuard lists, uBlock Origin lists, or
any other third-party filter list.

All rules are original to this project and MIT licensed under the terms in
`LICENSE`. The domain selections are informed by public knowledge of the
adtech and malicious-redirect ecosystems, but the rules themselves are not
copied from any third-party dataset.

## Why no third-party list is bundled

- **Licensing clarity.** EasyList is dual-licensed GPLv3 / CC BY-SA 3.0;
  AdGuard lists are LGPL-3.0 with additional terms. Redistributing converted
  rules can carry those obligations even though this extension's code is MIT.
- **Reproducible builds.** No dependency on an unpinned ABP-to-DNR converter
  whose output could change between builds.
- **Auditability.** 169 rules can be read end-to-end by one reviewer.

## Before adding a third-party list

Record in `docs/FILTER-SOURCES.md` **before** importing:

- Canonical HTTPS source URL
- SPDX license identifier and any additional terms
- Exact attribution text the license requires
- Retrieval date and SHA-256 of the fetched file
- Converter name, version, and configuration
- Whether the resulting DNR JSON may be redistributed

Redistributing derived rules may carry the source list's license obligations
even when the extension code is MIT. Treat list ingestion as a licensing and
supply-chain change, not a code change.

## Cross-ruleset duplication is intentional

Several domains appear in more than one ruleset (for example `criteo.com` in
both `ads.json` and `trackers.json`). This is deliberate: disabling one
ruleset must not silently remove protection for that domain. Do not "clean
up" cross-ruleset duplicates — only duplicates *within* a single ruleset are
bugs, and `npm run validate` fails on those.

## Trademark note

The project name and icon are not derived from uBlock Origin, AdGuard, or
EasyList. If any of those are ever forked or incorporated, keep their
trademarks out of the BuildMintZ Shield name and credit them explicitly here.

/* BuildMintZ Shield — options controller
 *
 * Every element lookup goes through requireEl(), so an HTML/JS ID mismatch
 * fails loudly with the missing ID in the console instead of silently
 * killing the rest of the script.
 */

'use strict';

const MAX_ALLOWLIST_ENTRIES = 4500;
const MAX_IMPORT_SIZE = 1024 * 1024;

const EXTENSION_ORIGIN = chrome.runtime.getURL('');

/* ---------- element access ---------- */

function requireEl(id) {
  const element = document.getElementById(id);

  if (!element) {
    throw new Error(`[BMZ options] missing element #${id}`);
  }

  return element;
}

/* ---------- helpers ---------- */

function normalizeHost(input) {
  if (typeof input !== 'string') return null;

  const value = input.trim();

  if (!value || /[\s/@*\\?#]/.test(value)) return null;

  try {
    const url = new URL(value.includes('://') ? value : `https://${value}`);

    if (!['http:', 'https:'].includes(url.protocol)) return null;
    if (url.username || url.password) return null;
    if (url.pathname !== '/' || url.search || url.hash) return null;

    const host = url.hostname
      .toLowerCase()
      .replace(/\.$/, '')
      .replace(/^www\./, '');

    if (
      !host.includes('.') ||
      host.length > 253 ||
      !/^[a-z0-9.-]+$/.test(host) ||
      host.startsWith('.') ||
      host.endsWith('.') ||
      host.includes('..')
    ) {
      return null;
    }

    return host;
  } catch {
    return null;
  }
}

function normalizeAllowlist(value) {
  if (!Array.isArray(value)) return [];

  return [...new Set(value.map(normalizeHost).filter(Boolean))]
    .slice(0, MAX_ALLOWLIST_ENTRIES);
}

function setMessage(text, isError = false) {
  const element = document.getElementById('message');
  if (!element) return;

  element.textContent = text || '';
  element.style.color = isError ? '#ff9a9a' : '#78e3bd';
}

/* ---------- allowlist store ---------- */

async function readAllowlist() {
  const stored = await chrome.storage.local.get({ allowlist: [] });
  return normalizeAllowlist(stored.allowlist);
}

async function writeAllowlist(next) {
  const clean = normalizeAllowlist(next);
  await chrome.storage.local.set({ allowlist: clean });
  return clean;
}

/* One serialized mutation queue. Rejections are swallowed for the chain but
 * still surfaced to the caller. */
let mutationTail = Promise.resolve();

function enqueueMutation(operation) {
  const next = mutationTail.then(operation);
  mutationTail = next.catch(() => {});
  return next;
}

function withAllowlist(handler) {
  return enqueueMutation(async () => {
    const current = await readAllowlist();
    return handler(current);
  });
}

function allowlistEquals(a, b) {
  return a.length === b.length && a.every((entry, index) => entry === b[index]);
}

/* ---------- render ---------- */

let rendering = false;

function hostFromUrl(url) {
  try {
    const parsed = new URL(url);
    if (!['http:', 'https:'].includes(parsed.protocol)) return '';
    return parsed.hostname.toLowerCase().replace(/^www\./, '');
  } catch {
    return '';
  }
}

function normalizeHost(value) {
  if (typeof value !== 'string') return '';
  return value.trim().toLowerCase().replace(/^www\./, '');

  const normalizedHost = normalizeHost(host);
  const normalizedEntry = normalizeHost(entry);

    normalizedHost &&
    normalizedEntry &&

# BuildMintZ Shield

A lightweight, local-only browser extension for blocking requests to known ad, tracking, and scam domains, and hiding common ad containers. It is available for Chrome, Microsoft Edge, and Firefox.

## What it does

- **Blocks Network Requests:** Blocks requests to ~155 hand-curated ad, tracker, and malicious-domain hosts using native `declarativeNetRequest` rules — no per-request JavaScript, no background CPU cost.
- **Cosmetic Filtering:** Hides common ad containers via a scoped cosmetic stylesheet plus a `MutationObserver` that catches dynamically inserted ads, including inline image overlays.
- **Video ad skipping:** On supported video-player pages, a lightweight content script clicks available skip controls and advances past client-side skippable pre-roll and mid-roll ads. *Note: server-stitched ads delivered inside the same media stream as the content cannot be skipped by any client-side extension; those will play normally.*
- **Dismisses client-side anti-adblock prompts:** When a supported player displays an "ad blockers are not allowed" message, the extension removes the prompt so the page continues to work.
- **Global Pause & Allowlist:** Offers a global pause switch and per-site exceptions directly from the popup.
- **Privacy First:** Does **not** require an account, subscription, telemetry endpoint, or any BuildMintZ-operated backend. All storage is local to your browser.
- **Ad-Slip Diagnostics:** When a video ad cannot be skipped within 1.5 seconds, the extension records the DOM signals it observed. This raw local log is viewable in the Settings page.

## What it does not do

- **It is not undetectable:** Sites can always detect client-side page modification. The extension is designed to fail safely — if it encounters a page it cannot handle, the page continues to work normally and ads simply remain visible.
- **It cannot block server-stitched video ads:** Ads delivered inside the same media stream as the content cannot be skipped or blocked by any client-side extension. The player is not broken; those specific ads will just play normally.
- **Not an antivirus:** The security rule group is small and hand-authored. Keep your browser's Safe Browsing enabled.
- **No remote filter lists:** Rules ship with this release. A future updater would need signed or pinned provenance — see `docs/FILTER-SOURCES.md`.

6. Pin the extension to your toolbar and open its popup to confirm the protection status.

const DOS_TIME = 0;
const DOS_DATE = 0x21; // 1980-01-01

function crc32(buffer) {
  let table = crc32.table;

  if (!table) {
    table = crc32.table = new Uint32Array(256);

    for (let i = 0; i < 256; i++) {
      let value = i;

      for (let bit = 0; bit < 8; bit++) {
        value = value & 1
          ? (value >>> 1) ^ 0xEDB88320
          : value >>> 1;
      }

      table[i] = value >>> 0;
    }
  }

  let crc = 0xFFFFFFFF;

  for (const byte of buffer) {
    crc = table[(crc ^ byte) & 0xFF] ^ (crc >>> 8);
  }

  return (crc ^ 0xFFFFFFFF) >>> 0;
}

function collectFiles(directory) {
  const files = [];

  function walk(currentDirectory, prefix = '') {
    const entries = fs.readdirSync(currentDirectory, {
      withFileTypes: true
    }).sort((a, b) => a.name.localeCompare(b.name, 'en'));

    for (const entry of entries) {
      const absolute = path.join(currentDirectory, entry.name);
      const archiveName = `${prefix}${entry.name}`;

      if (entry.isSymbolicLink()) {
        throw new Error(`Symbolic links are not allowed in packages: ${absolute}`);
      }

      if (entry.isDirectory()) {
        walk(absolute, `${archiveName}/`);
        continue;
      }

      if (!entry.isFile()) {
        throw new Error(`Unsupported file type in package: ${absolute}`);
      }

      files.push({
        name: archiveName.split(path.sep).join('/'),
        data: fs.readFileSync(absolute)
      });
    }
  }

  walk(directory);
  return files;
}

function zipDirectory(directory, outputFile) {
  const files = collectFiles(directory);

  const localChunks = [];
  const centralChunks = [];
  let offset = 0;

  if (files.length > 0xFFFF) {
    throw new Error('Too many files for the ZIP format supported by this builder.');
  }

  for (const file of files) {
    const name = Buffer.from(file.name, 'utf8');
    const original = file.data;

    if (name.length > 0xFFFF) {
      throw new Error(`ZIP filename is too long: ${file.name}`);
    }

    const compressed = zlib.deflateRawSync(original, {
      level: 9
    });

    const checksum = crc32(original);

    if (
      original.length > 0xFFFFFFFF ||
      compressed.length > 0xFFFFFFFF ||
      offset > 0xFFFFFFFF
    ) {
      throw new Error('ZIP64 is not supported by this builder.');
    }

    const localHeader = Buffer.alloc(30);
    localHeader.writeUInt32LE(0x04034B50, 0);
    localHeader.writeUInt16LE(20, 4);
    localHeader.writeUInt16LE(0x0800, 6);
    localHeader.writeUInt16LE(8, 8);
    localHeader.writeUInt16LE(DOS_TIME, 10);
    localHeader.writeUInt16LE(DOS_DATE, 12);
    localHeader.writeUInt32LE(checksum, 14);
    localHeader.writeUInt32LE(compressed.length, 18);
    localHeader.writeUInt32LE(original.length, 22);
    localHeader.writeUInt16LE(name.length, 26);
    localHeader.writeUInt16LE(0, 28);

    localChunks.push(localHeader, name, compressed);

    const centralHeader = Buffer.alloc(46);
    centralHeader.writeUInt32LE(0x02014B50, 0);
    centralHeader.writeUInt16LE(20, 4);
    centralHeader.writeUInt16LE(20, 6);
    centralHeader.writeUInt16LE(0x0800, 8);
    centralHeader.writeUInt16LE(8, 10);
    centralHeader.writeUInt16LE(DOS_TIME, 12);
    centralHeader.writeUInt16LE(DOS_DATE, 14);
    centralHeader.writeUInt32LE(checksum, 16);
    centralHeader.writeUInt32LE(compressed.length, 20);
    centralHeader.writeUInt32LE(original.length, 24);
    centralHeader.writeUInt16LE(name.length, 28);
    centralHeader.writeUInt16LE(0, 30);
    centralHeader.writeUInt16LE(0, 32);
    centralHeader.writeUInt16LE(0, 34);
    centralHeader.writeUInt16LE(0, 36);
    centralHeader.writeUInt32LE(0, 38);
    centralHeader.writeUInt32LE(offset, 42);

    centralChunks.push(centralHeader, name);

    offset += localHeader.length + name.length + compressed.length;
  }

  const centralDirectory = Buffer.concat(centralChunks);
  const centralOffset = offset;

  if (centralDirectory.length > 0xFFFFFFFF) {
    throw new Error('ZIP central directory exceeds the supported size.');
  }

  const endRecord = Buffer.alloc(22);
  endRecord.writeUInt32LE(0x06054B50, 0);
  endRecord.writeUInt16LE(0, 4);
  endRecord.writeUInt16LE(0, 6);
  endRecord.writeUInt16LE(files.length, 8);
  endRecord.writeUInt16LE(files.length, 10);
  endRecord.writeUInt32LE(centralDirectory.length, 12);
  endRecord.writeUInt32LE(centralOffset, 16);
  endRecord.writeUInt16LE(0, 20);

  fs.writeFileSync(
    outputFile,
    Buffer.concat([...localChunks, centralDirectory, endRecord])
  );
}

    .sort((a, b) => a.archiveName.localeCompare(b.archiveName))

  if (buildFirefox) {
    console.log(
      'Firefox: test the generated package in the intended Firefox version before distribution.'
    );
  }

function validateAction(action, label) {
  assert(isPlainObject(action), `${label}: action must be an object`);
  if (!isPlainObject(action)) return;

  for (const key of Object.keys(action)) {
    if (!VALID_ACTION_KEYS.has(key)) {
      warn(`${label}: unknown action key "${key}"`);
    }
  }

  if (!assert(VALID_ACTION_TYPES.has(action.type), `${label}: unsupported action type "${action.type}"`)) {
    return;
  }

  if (action.type === 'redirect') {
    const redirect = action.redirect;
    assert(
      isPlainObject(redirect) &&
        (typeof redirect.extensionPath === 'string' ||
          typeof redirect.url === 'string' ||
          typeof redirect.regexSubstitution === 'string' ||
          isPlainObject(redirect.transform)),
      `${label}: redirect action requires extensionPath, url, regexSubstitution, or transform`
    );
  }

  if (action.type === 'modifyHeaders') {
    assert(
      Array.isArray(action.requestHeaders) || Array.isArray(action.responseHeaders),
      `${label}: modifyHeaders requires requestHeaders or responseHeaders`
    );
  }
}

function validateCondition(condition, label, rulesetId) {
  assert(isPlainObject(condition), `${label}: condition must be an object`);
  if (!isPlainObject(condition)) return null;

  for (const key of Object.keys(condition)) {
    if (!VALID_CONDITION_KEYS.has(key)) {
      warn(`${label}: unknown condition key "${key}"`);
    }
  }

  const hasUrlFilter = typeof condition.urlFilter === 'string' && condition.urlFilter.length > 0;
  const hasRegexFilter = typeof condition.regexFilter === 'string' && condition.regexFilter.length > 0;

  assert(hasUrlFilter || hasRegexFilter, `${label}: needs a non-empty urlFilter or regexFilter`);

  if (hasUrlFilter) {
    if (/\s/.test(condition.urlFilter)) {
      fail(`${label}: urlFilter contains whitespace, which DNR rejects`);
    }

    assert(condition.urlFilter.length <= 2000, `${label}: urlFilter exceeds 2000 characters`);
  }

  if (condition.resourceTypes !== undefined) {
    validateResourceTypes(condition.resourceTypes, label);

    if (
      Array.isArray(condition.resourceTypes) &&
      condition.resourceTypes.includes('main_frame') &&
      !RULESETS_ALLOWED_MAIN_FRAME.has(rulesetId)
    ) {
      warn(`${label}: main_frame is blocked from ruleset "${rulesetId}" — confirm this is intentional`);
    }
  }

  if (condition.excludedResourceTypes !== undefined) {
    validateResourceTypes(condition.excludedResourceTypes, label);
  }

  for (const key of [
    'initiatorDomains',
    'excludedInitiatorDomains',
    'requestDomains',
    'excludedRequestDomains',
    'domains',
    'excludedDomains'
  ]) {
    if (condition[key] !== undefined) validateDomains(condition[key], label, key);
  }

  assert(
    condition.urlFilter === undefined || condition.regexFilter === undefined,
    `${label}: urlFilter and regexFilter are mutually exclusive`
  );

  return hasUrlFilter ? condition.urlFilter : null;
}

function validateRule(rule, rulesetId, seenIds, seenFilters) {
  const label = `${rulesetId} rule ${String(rule?.id ?? '?')}`;

  if (!assert(isPlainObject(rule), `${label}: rule must be an object`)) return;

  assert(
    Number.isInteger(rule.id) && rule.id >= 1 && rule.id < MAX_RULE_ID,
    `${label}: id must be a positive integer below ${MAX_RULE_ID}`
  );

  if (Number.isInteger(rule.id)) {
    if (seenIds.has(rule.id)) {
      fail(`${label}: duplicate rule ID`);
    } else {
      seenIds.add(rule.id);
    }
  }

  if (rule.priority !== undefined) {
    assert(
      Number.isInteger(rule.priority) && rule.priority >= 1,
      `${label}: priority must be a positive integer`
    );
  }

  validateAction(rule.action, label);

  const urlFilter = validateCondition(rule.condition, label, rulesetId);

  if (typeof urlFilter === 'string') {
    if (seenFilters.has(urlFilter)) {
      fail(`${label}: duplicate urlFilter "${urlFilter}" within ruleset "${rulesetId}"`);
    } else {
      seenFilters.add(urlFilter);
    }
  }
}

    'manifest.version must be a dotted numeric version'
  );

  assert(
    isPlainObject(manifest.background) && typeof manifest.background.service_worker === 'string',
    'background.service_worker is required'
  );

  assert(Array.isArray(manifest.permissions), 'permissions must be an array');
  assert(
    manifest.permissions?.includes('declarativeNetRequest'),
    'declarativeNetRequest permission is required'
  );

  assert(Array.isArray(manifest.host_permissions), 'host_permissions must be an array');
  assert(
    manifest.host_permissions?.includes('http://*/*'),
    'http://*/* host permission is required'
  );
  assert(
    manifest.host_permissions?.includes('https://*/*'),
    'https://*/* host permission is required'
  );

  for (const size of ['16', '48', '128']) {
    const iconPath = manifest.icons?.[size];

    if (typeof iconPath !== 'string' || !iconPath) {
      fail(`manifest.icons["${size}"] is required`);
    } else {
      exists(iconPath);
    }
  }

  if (manifest.background?.service_worker) exists(manifest.background.service_worker);
  if (manifest.action?.default_popup) exists(manifest.action.default_popup);
  if (manifest.options_page) exists(manifest.options_page);
  if (manifest.options_ui?.page) exists(manifest.options_ui.page);

  for (const contentScript of manifest.content_scripts ?? []) {
    assert(
      Array.isArray(contentScript.matches) && contentScript.matches.length > 0,
      'content_scripts[].matches must be a non-empty array'
    );

    if (contentScript.all_frames !== undefined) {
      assert(
        typeof contentScript.all_frames === 'boolean',
        'content_scripts[].all_frames must be a boolean'
      );
    }

    for (const file of contentScript.js ?? []) exists(file);
    for (const file of contentScript.css ?? []) exists(file);
  }

  const resources = manifest.declarative_net_request?.rule_resources;

  if (!assert(Array.isArray(resources) && resources.length > 0, 'at least one rule resource is required')) {
    return manifest;
  }

  assert(resources.length <= MAX_STATIC_RULESETS, `declared static rulesets exceed ${MAX_STATIC_RULESETS}`);

  const enabledCount = resources.filter(resource => resource?.enabled === true).length;
  assert(
    enabledCount <= MAX_ENABLED_STATIC_RULESETS,
    `enabled static rulesets (${enabledCount}) exceed ${MAX_ENABLED_STATIC_RULESETS}`
  );

  const rulesetIds = new Set();
  const rulesetPaths = new Set();
  let totalRules = 0;

    if (!isPlainObject(resource)) {
      fail('each rule resource must be an object');
      continue;
    }

    assert(typeof resource.id === 'string' && resource.id.length > 0, 'ruleset id is required');
    assert(typeof resource.path === 'string' && resource.path.length > 0, `ruleset ${resource.id || '?'}: path is required`);
    assert(typeof resource.enabled === 'boolean', `ruleset ${resource.id || '?'}: enabled must be a boolean`);

    if (typeof resource.id !== 'string' || !resource.id) continue;

    if (rulesetIds.has(resource.id)) fail(`duplicate ruleset id: ${resource.id}`);
    rulesetIds.add(resource.id);

    if (typeof resource.path !== 'string' || !resource.path) continue;

    if (rulesetPaths.has(resource.path)) fail(`duplicate ruleset path: ${resource.path}`);
    rulesetPaths.add(resource.path);

    const absolute = path.join(ROOT, resource.path);

    if (fs.existsSync(absolute)) {
      const bytes = fs.statSync(absolute).size;

      if (bytes > MAX_RULE_FILE_BYTES) {
        fail(`${resource.path} is ${(bytes / 1024 / 1024).toFixed(1)} MB, above the 2 MB cap`);
      }
    }

    const rules = loadJSON(resource.path);

    if (!Array.isArray(rules)) {
      fail(`${resource.path} must contain a JSON array`);
      continue;
    }

    assert(
      rules.length <= MAX_RULES_PER_RULESET,
      `${resource.path}: ${rules.length} rules exceeds the per-ruleset cap of ${MAX_RULES_PER_RULESET}`
    );

    const seenFilters = new Set();

    for (const rule of rules) {
      validateRule(rule, resource.id, seenIds, seenFilters);
    }

    totalRules += rules.length;
    console.log(`  ${resource.id}: ${rules.length} rules`);
  }

  assert(totalRules <= MAX_TOTAL_RULES, `total static rules ${totalRules} exceeds the cap of ${MAX_TOTAL_RULES}`);
  console.log(`\nTotal static rules: ${totalRules}`);

}

/* Cross-check every element ID referenced by a controller script against
 * the IDs actually defined in its HTML. This is the check that would have
 * caught the four broken references on the options page in one second. */
function validateElementReferences() {
  const pairs = [
    { js: 'popup/popup.js', html: 'popup/popup.html' },
    { js: 'options/options.js', html: 'options/options.html' }
  ];

  for (const { js, html } of pairs) {
    const script = readText(js);
    const markup = readText(html);

    if (script === null || markup === null) continue;

    const referenced = new Set(
      [...script.matchAll(/(?:\$|requireEl|getElementById)\(\s*'([A-Za-z][\w-]*)'\s*\)/g)].map(
        match => match[1]
      )
    );

    const defined = new Set(
      [...markup.matchAll(/\bid="([^"]+)"/g)].map(match => match[1])
    );

    for (const id of referenced) {
      if (!defined.has(id)) {
        fail(`${js} references #${id} but ${html} does not define it`);
      }
    }
  }
}

/* Keep the rule counts quoted in docs/SECURITY-MODEL.md from drifting away
 * from the actual JSON. If the sentence is reworded, warn instead of fail. */
function validateDocumentedCounts(manifest, counts) {
  const doc = readText('docs/SECURITY-MODEL.md');
  if (doc === null) return;

  const match = doc.match(
    /(\d+)\s+known ad domains,\s*(\d+)\s+tracker domains,\s*and\s*(\d+)\s+confirmed/i
  );

  if (!match) {
    warn('could not find the rule-count sentence in docs/SECURITY-MODEL.md');
    return;
  }

  const documented = {
    ads: Number(match[1]),
    trackers: Number(match[2]),
    security: Number(match[3])
  };

  const byRuleset = {};
  for (const resource of manifest?.declarative_net_request?.rule_resources ?? []) {
    const rules = loadJSON(resource.path);
    if (Array.isArray(rules)) byRuleset[resource.id] = rules.length;
  }

  for (const key of Object.keys(documented)) {
    if (byRuleset[key] !== documented[key]) {
      fail(
        `docs/SECURITY-MODEL.md claims ${documented[key]} ${key} rules but ` +
          `${key}.json contains ${byRuleset[key] ?? 'an unknown number of'} rules`
      );
    }
  }

  void counts;
}

}
